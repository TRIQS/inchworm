\section{Expansion of the propagator}

This proof is similar to the one presented in the review~\cite{gull_continuous-time_2011}.
Let's start from imaginary time evolution operator defined as (FIXME: add proof of this definition):
\begin{align}
U(\tau)=& \frac{1}{\ZB} \operatorname{Tr_B} [e^{-(\beta-\tau) H_B } e^{-\tau H }] 
\\
\ZB =& \operatorname{Tr_B} [e^{-\beta H_B }] 
\end{align}

In interaction representation, where $H=H_B+H_\text{at}+H_\text{hyb}$. The expansion is done on $H_\text{hyb}$ so $H_0 = H_\text{at} + H_\text{B}$
\begin{align}
U(\tau)=& \frac{1}{\ZB} \operatorname{Tr_B} [e^{-(\beta - \tau) H_B } e^{-\tau (H_B + H_{\mathrm{at}} + H_\text{hyb}) }]  
\\ 
=&  \frac{1}{\ZB} 
\sum_{k=0}^{\infty} \int_{0}^{\tau} d \tau_{1} \cdots \int_{\tau_{k-1}}^{\tau} d \tau_{k} \int_{0}^{\tau} d \tau_{1}^{\prime} \cdots \int_{\tau_{k^{\prime}-1}}^{\tau} d \tau_{k}^{\prime} 
\nonumber
\\
& \times \operatorname{Tr_B}\left[T_{\tau} e^{-(\beta-\tau) H_\mathrm{B}} e^{-\tau (H_\mathrm B + H_{\mathrm{at}})} \tilde{H}_{\mathrm{hyb}}\left(\tau_{k}\right) \tilde{H}_{\mathrm{hyb}}^{\dagger}\left(\tau_{k}^{\prime}\right) \cdots\right. \left.\times \tilde{H}_{\mathrm{hyb}}\left(\tau_{1}\right) \tilde{H}_{\mathrm{hyb}}^{\dagger}\left(\tau_{1}^{\prime}\right)\right]
\\ 
=&  \frac{1}{\ZB} 
\sum_{k=0}^{\infty} \int_{0}^{\tau} d \tau_{1} \cdots \int_{\tau_{k-1}}^{\tau} d \tau_{k} \int_{0}^{\tau} d \tau_{1}^{\prime} \cdots \int_{\tau_{k^{\prime}-1}}^{\tau} d \tau_{k}^{\prime} 
\nonumber
\\
& \times \operatorname{Tr_B}\left[T_{\tau} e^{-\beta H_\mathrm{B}} e^{-\tau H_{\mathrm{at}}} \tilde{H}_{\mathrm{hyb}}\left(\tau_{k}\right) \tilde{H}_{\mathrm{hyb}}^{\dagger}\left(\tau_{k}^{\prime}\right) \cdots\right. \left.\times \tilde{H}_{\mathrm{hyb}}\left(\tau_{1}\right) \tilde{H}_{\mathrm{hyb}}^{\dagger}\left(\tau_{1}^{\prime}\right)\right]
\end{align}

where
$H_{\mathrm{hyb}}=\sum_{p j}\left(\theta_{p}^{j} c_{p}^{\dagger} d_{j}+\theta_{p}^{j *} d_{j}^{\dagger} c_{p}\right)=\tilde{H}_{\mathrm{hyb}}+\tilde{H}_{\mathrm{hyb}}^{\dagger}$. The trace over the baths have for consequence that only the terms with same numbers of $\tilde{H}_{\mathrm{hyb}}$ and $\tilde{H}_{\mathrm{hyb}}^{\dagger}$ survive in the sum above. This results in a $(-1)^2$ for the only surviving terms of the sum, thus the absence of minus sign. In the last equality, splitting the exponential in two is possible here because $[H_{\mathrm{at}},H_{\mathrm{B}}]=0$.
After that: 
\begin{align}
U(\tau)=& \frac{1}{\ZB}\sum_{k=0}^{\infty}   \int_{0}^{\tau} d \tau_{1} \cdots \int_{\tau_{k-1}}^{\tau} d \tau_{k} \int_{0}^{\tau} d \tau_{1}^{\prime} \cdots \int_{\tau_{k-1}^{\prime}}^{\tau} d \tau_{k}^{\prime} 
\sum_{j_1^{(\prime)} \cdots j_k^{(\prime)}} \sum_{p_1^{(\prime)} \cdots  p_k^{(\prime)}} 
\theta_{p_{1}}^{j_{1}} \theta_{p_{1}^{\prime}}^{j_{1}^{\prime}*} \cdots \theta_{p_{k}}^{j_{k}} \theta_{p_{k}^{\prime}}^{j_k^{\prime}*} 
\nonumber
\\
& \times \operatorname{Tr_B}\left[T_{\tau} e^{-\beta H_\mathrm{B}} e^{-\tau H_{\mathrm{at}}} d_{j_{k}}\left(\tau_{k}\right) c_{p_{k}}^{\dagger}\left(\tau_{k}\right) c_{p_{k^{\prime}}}\left(\tau_{k}^{\prime}\right)   d_{j_{k}^{\prime}}^{\dagger}\left(\tau_{k}^{\prime}\right)
d_{j_{1}}\left(\tau_{1}\right) c_{p_{1}}^{\dagger}\left(\tau_{1}\right) c_{p_{1}^{\prime}}\left(\tau_{1}^{\prime}\right) d_{j_{1}^{\prime}}^{\dagger}\left(\tau_{1}^{\prime}\right)\right]
\\
=& \frac{1}{\ZB}\sum_{k=0}^{\infty}  \int_{0}^{\tau} d \tau_{1} \cdots \int_{\tau_{k-1}}^{\tau} d \tau_{k} \int_{0}^{\tau} d \tau_{1}^{\prime} \cdots \int_{\tau_{k-1}^{\prime}}^{\tau} d \tau_{k}^{\prime}
\sum_{j_1^{(\prime)} \cdots j_k^{(\prime)}} \sum_{p_1^{(\prime)} \cdots  p_k^{(\prime)}} 
\theta_{p_{1}}^{j_{1}} \theta_{p_{1}^{\prime}}^{j_{1}^{\prime}*} \cdots \theta_{p_{k}}^{j_{k}} \theta_{p_{k}^{\prime}}^{j_k^{\prime}*} 
\nonumber
\\
& \times \Big[T_{\tau} e^{-\beta H_{\mathrm{H_\mathrm{B}}}} d_{j_{k}}\left(\tau_{k}\right) d_{j_{k}^{\prime}}^{\dagger}\left(\tau_{k}^{\prime}\right) \cdots d_{j_{1}}\left(\tau_{1}\right) d_{j_{1}^{\prime}}^{\dagger}\left(\tau_{1}^{\prime}\right)\Big]
\operatorname{Tr_B}
\Big[ T_{\tau} e^{-\beta H_{\mathrm{B}}} c_{p_{k}}^{\dagger}\left(\tau_{k}\right) c_{p_{k^{\prime}}}\left(\tau_{k}^{\prime}\right) \cdots c_{p_{1}}^{\dagger}\left(\tau_{1}\right) c_{p_{1}^{\prime}}\left(\tau_{1}^{\prime}\right)\Big]
\label{eq::u2}
\end{align}


We define the hybridization partition of the bath (with $k$ electron) as:
\begin{align}
Z_{\mathrm{B}}=\operatorname{Tr}_\mathrm B e^{-\beta H_{\mathrm{B}}}=\prod_{\sigma} \prod_{p}^k \left(1+e^{-\beta \epsilon_{p}}\right)
\end{align}

For every bath $p$, the energy is given by:
\begin{align}
\Delta_{l m}(\tau)=\sum_{p} \frac{\theta_{p}^{l *} \theta_{p}^{m}}{e^{\epsilon_{p} \beta}+1} \times\left\{\begin{array}{ll}
{-e^{-\epsilon_{p}(\tau-\beta)},} & {0<\tau<\beta} \\
{e^{-\epsilon_{p} \tau},} & {-\beta<\tau<0}
\end{array}\right.
\end{align}

Then:
\begin{align}
\label{eq_det_Hyb}
\frac{1}{\ZB} \operatorname{Tr_B}\Big[T_{\tau} e^{-\beta H_{\text {B }}} \sum_{p_1 \ldots p_k}  \sum_{p_{1}^{\prime} \ldots p_k^{\prime}} 
\theta_{p_{1}}^{j_{1}} \theta_{p_{1}^{\prime}}^{j_{1}^{\prime} *} \cdots \theta_{p_{k}}^{j_{k}} \theta_{p_{k}^{\prime}}^{j_{k}^{\prime}*}  
c_{p_{k}}^{\dagger}\left(\tau_{k}\right) c_{p_{k^{\prime}}}\left(\tau_{k}^{\prime}\right) \cdots c_{p_{1}}^{\dagger}\left(\tau_{1}\right) c_{p_{1}^{\prime}}\left(\tau_{1}^{\prime}\right)\Big]=\operatorname{det} \Delta
\end{align}

where the hybridization matrix $\Delta$ is of the form:
\begin{equation}
\Delta = \left(\begin{array}{ccc}{
\Delta_{p_1 p_1^\prime}(\tau_{1}^{\prime} - \tau_{1})} &  
{\cdots} & 
{\Delta_{p_1 p_k^\prime}(\tau_{k}^{\prime} - \tau_{1})}  
\\ {\vdots} & {\ddots} & {\vdots} 
\\ 
{\Delta_{p_k p_1^\prime}(\tau_{1}^{\prime} - \tau_{k})} & 
{\cdots} & 
{\Delta_{p_k p_k^\prime}(\tau_{k}^{\prime} - \tau_{k} )}
\end{array}\right)
\label{eq::hyb_mat}
\end{equation}
%
The Leibniz formula for the determinant of an $\Delta$ matrix (size $n \times n$, here $n=k$, but the notation $S_n$ usually denote the permutation group of size $n$) is:
\begin{equation}
\operatorname{det} \Delta=\sum_{\sigma \in S_n} \operatorname{sgn}(\sigma) 
\prod_{i=1} \Delta_{p_{i} p^\prime_{\sigma_i}}
(\tau_{\sigma_i}^{\prime} - \tau_{i})
\label{eq::Leibniz_det}
\end{equation}
where $\sigma$ is permutation  and $\operatorname{sgn}(\sigma)$ is the sign of the permutation. Using Eq.~\eqref{eq_det_Hyb} and~\eqref{eq::u2}
, we then have:
\begin{align}
U(\tau)=& \sum_k^{\infty}
\int_{0}^{\tau} d \tau_1 \cdots \int_{\tau_{k-1}}^{\tau} d \tau_k
\int_{0}^{\tau} d \tau_1^\prime \cdots \int_{\tau_{k-1}^\prime}^{\tau} d \tau_k^{\prime}
\sum_{j_1 \ldots j_k}  \sum_{j_{1}^{\prime} \ldots j_k^{\prime}} 
\Big[ T_{\tau} e^{-\beta H_\mathrm{at}}
d_{j_k}(\tau_k) d_{j_k^\prime}^{\dagger} (\tau_k^\prime) \cdots d_{j_1}(\tau_1) d_{j_1^\prime}^{\dagger} (\tau_1^\prime) \Big] \operatorname{det} \Delta
\label{eq::normal_order}
\end{align}
%
where $\tau_k > \cdots >\tau_1$ and $\tau_k^\prime > \cdots >\tau_1^\prime$. This is the normal order, i.e. the order that does not have a sign. Practically, we need to order the operators for all the times togheter such that $\tau_{{2k}} > \tau_{{2k-1}} > \cdots >\tau_{{1}}$. This renaming of the times correspond do zip both $\tau_k$ and $\tau^\prime_k$. This reordering imply a new sign $(-1)^p$ where $p$ is the number of permutation one must do to order all the times starting from the normal order. 

Using the bare time evolution $U_0(\tau) = e^{-\tau H_{\mathrm{at}}}$ to express the time of the operator $d_{j_k^\prime}^{\dagger}(\tau_k^\prime)$ and $d_{j_k}(\tau_k)$, and using the notation $D_a$ to refer to the first $d$ or $d^\dagger$, $D_b$ the second and so on:
\begin{align}
U(\tau)=& \sum_k^{\infty}
\int \prod_i^{2k} \Big( d \tau_{i} \Big)  \sum_{\{a_i\}} 
  \Big[ (-1)^p 
U_0(\tau-\tau_{a_{2k}}) D_{a_{2k}}
U_0(\tau_{a_{2k}}-\tau_{a_{2k-1}}) D_{a_{2k-1}}
\cdots 
U_0(\tau_{a_2}-\tau_{a_1}) D_{a_1}
U_0(\tau_{a_{1}})
\Big] \operatorname{det} \Delta
\label{eq::ordered_order}
\end{align}

This equation can be rewritten in a more compact form:
\begin{align}
U(\tau)=& 
\sum_k^{\infty} \int \prod_i^{2k} \Big( d \tau_{i} \Big)  \sum_{\{a_i\}} 
w_{\mathrm{at}} w_{\mathrm{hyb}}
\end{align}
%
%
where the weights are:
\begin{align}
w_{\mathrm{at}} =&
 (-1)^\sigma 
U_0(\tau-\tau_{a_{2k}}) D_{a_{2k}}
U_0(\tau_{a_{2k}}-\tau_{a_{2k-1}}) D_{a_{2k-1}}
\cdots 
U_0(\tau_{a_2}-\tau_{a_1}) D_{a_1}
U_0(\tau_{a_{1}})
\\
w_{\mathrm{hyb}} =&
\operatorname{det} \Delta
\label{eq::ordered_order}
\end{align}

\subsection{Specific to my implementation}
There are many ways to evaluate this $\sigma$. In my implementation, the $d^\dagger$ and $d$ are stored in two vectors, while the the full time-ordered list of operators (including both  $d^\dagger$ and $d$) is stored in another vector. It is more natural to work with vector where the time is sorted in the ascending order. This is also the order in which we apply the operator in the matrix products. In that case, the number of permutation between the normal order and the sorted order ($\sigma$) is simply the sum of the positions of the $d^\dagger$ operators in the sorted order. Suppose for example that after ordering all the times, you get the product: $d^\dagger_0 d_1 d_2 d_3^\dagger d_4 d_5^\dagger$, the reordering produce the sign $(-1)^{0+3+5}=1$




\section{Inchworm}

Up until now, we talked about the propagator, but mainly in the context of CT-HYB, i.e. normal sampling of the diagrams. The inchworm consist of taking Eq.~\eqref{eq::ordered_order} and splitting at time $\tau_{split}$. Doing so, you can use the full propagator $U(\tau)$ and only consider the connected hybridization terms. Hence the atomic weights become:
\begin{align}
w_{\mathrm{at}} =&
(-1)^p
U(\tau-\tau_{a_{2k}}) D_{a_{2k}}
U(\tau_{a_{2k}}-\tau_{a_{2k-1}}) D_{a_{2k-1}}
\cdots   
\label{eq::ordered_order}
\nonumber
\\
&
\cdots 
U(\tau_{n+1}-\tau_{split}) 
U(\tau_{split}-\tau_{n})  
\cdots 
U(\tau_{a_2}-\tau_{a_1}) D_{a_1}
U(\tau_{a_{1}})
\end{align}
%
The hybridization weight $w_{hyb}$ is similar to Eq.~\eqref{eq::Leibniz_det} where we consider only the proper diagram of this summation:
\begin{equation}
w_\text{hyb} =
\sum_{\substack{\sigma \in S_n \\ \text{proper}}} \operatorname{sgn}(\sigma) 
\prod_{i=1} \Delta_{p_{i} p^\prime_{\sigma_i}}
(\tau_{\sigma_i}^{\prime} - \tau_{i})
\label{eq::w_hyb}
\end{equation}

\subsection{Green function}

In the inchworm algorithm, the Green function must be calculated once you have the full propagator $U(\tau)$ for $0 < \tau < \beta$. Of course you only have a discrete mesh, but you can interpolate. The Green function then reads:

\begin{align}
G_{a b}(\tau_{split})=& \operatorname{Tr} \bigg[
\sum_k^{\infty} \int \prod_i^{2k} \Big( d \tau_{i} \Big)  \sum_{\{a_i\}} 
w^{ab}_{\mathrm{at}} w_{\mathrm{hyb}}
\bigg] /   \operatorname{Tr} \big[  U(\beta) \big]
\end{align}

where $w^{ab}_{\mathrm{at}}$ is similar to $w_{\mathrm{at}}$ but with an additionnal $d_a$ at imaginary time $0$ and a $d^{\dagger}_b$ at time $t_{split}$:

d_a
\end{align}

\section{Hybridization weight}
\label{sec::w_hyb}
This section discuss about $w_{hyb}$ as defined in Eq.~\eqref{eq::w_hyb}.

In the following two sections, we will explore the formalism used here along a simple example that can be generated with the code \url{https://github.com/TRIQS/inchworm.git} under the tag \Verb#doc_0.1# (commit number \Verb#d0745e5#). After building with \Verb#cmake#, run in the file \Verb#build_directory/test/c++/diagram_analysis# to generate the example below.

We explore two approaches. The first approach, proper diagram (Sec~\ref{sec::proper}), consist of enumerating all the $k!$ diagrams (where $k$ is the order of expansion) and determine if they are proper (fully-connected). This section also define clearly what is this criterion. The second approach, inclusion-exclusion (Sec~\ref{sec::inclu}), is more elaborated and complicated, but it result in a much faster algorithm.

\subsection{Definitions}

\label{sec::definition}

%Suppose a time $\tau_{split}$, 

An hybridization line is considered crossing with $\tau_{c}$ if $\Delta_{ab}(\tau,\tau^\prime)$ has $\tau < \tau_{c} < \tau^\prime$ or  $\tau > \tau_{c} > \tau^\prime$. An hybridization line is considered connected if it is crossing with either a split time $\tau_{split}$ or one and only one of the two times ($\tau$ and $\tau^\prime$) of at least one other connected hybridization line. And a diagram is considered proper if every of its hybridization are connected to a split time $\tau_{split}$.

Note that this definition of connectivity is different from the one defined usually in the inchworm reference papers~\cite{boag_inclusion-exclusion_2018,antipov_currents_2017,eidelstein_multiorbital_2019}. This is related to the fact that Eq.~\eqref{eq::inchworm_order} is also a little different than the one found in~\cite{antipov_currents_2017}, since we use full propagator on both sides of $\tau_{split}$. Hence not every point on the right side of the split point are connected by default. This will result in algorithm that are slightly different for both ``proper diagrams enumaration'' and  ``inclusion-exclusion'' methods presented in the next sections.


\subsection{Proper diagrams enumaration}

\label{sec::proper}

What we call a diagram here is one specific way to reconnect all the $d(\tau)$ and $d^\dagger(\tau^\prime)$. Note that there is one hybridization $\Delta_{ab}(\tau,\tau^\prime)$  per $d(\tau)$ and $d^\dagger(\tau^\prime)$, i.e. as much as the order of expansion $k$. Also, there $k!$ way to connect a specific configuration $d(\tau)$ and $d^\dagger(\tau^\prime)$ with hybridization lines, hence $k!$ different diagrams to analyse. For the example here $k=4$ hence there is  $24$ different configurations to consider. This get huge fast.

To find if a diagram is proper, we need to implement a simple depth first search algorithm. We start by listing all $\Delta_{ab}(\tau,\tau^\prime)$  that cross $\tau_{split}$, and from these, successivly search all the $\Delta_{ab}(\tau,\tau^\prime)$ that are connected. This deep first search algorithm is simple, but let's consider an example. Let's consider that after ordering in time the creation an annihilaiton operator, we obtain the chain $d(\tau_0)d^\dagger(\tau_1)d(\tau_2)d^\dagger(\tau_3)d^\dagger(\tau_4)d(\tau_5)d^\dagger(\tau_6)d(\tau_7)$ where time are ordered ($\tau_i<\tau_{i+1}$).  Only the order of operator matter here, so this information can be summarized with the diagram like \Verb#x-o-x-o-o-x-o-x#. 

Now suppose a split time $\tau_{split}$ such that $\tau_5 < \tau_{split} < \tau_6$. We need to find, for a distribution of hybridization line if a diagram is proper or not.  Let's take and analyse one of these diagrams. The vector of \Verb#segment_list# contains all the single segments possible on this diagram. A \Verb#segment# must contain the same number of \Verb#o# (creation operator) and \Verb#x# (annihilation operator), and it must not cross a split point (vertical line on the graph). \Verb#===# lines represent segments and \Verb#---# lines represent the rest of the diagram that is not segment.

From this list we can generate the two other lists by combining every possible segment, successively. If they are all adjacent, it is pushed back in \Verb#list_of_set_of_adjacent_segments# and if they are all disjoint, it is pushed back in the vector of \Verb#set_of_disjoint_segments#, otherwise, it is forgotten. These three vectors are necessary and sufficient to calculate the value associated with every single \Verb#segment#.

We start by calculating the value of the smaller segments as any \Verb#segment# depends only on the value of \Verb#segment# smaller than itself, and the smallest segments are trivial to calculate. 

The value of a single \Verb#segment# ($S$) can be calculated in two steps, first calculate the value without adjacent segment (or value without cuts in the source code), noted $c_k(V)$ (or \Verb#+++# sign in the graphs below) and the value with adjacent segments (or value in source code), noted $a_k(S)$ (or \Verb#===# sign in the graphs below) from these two equations:
\begin{align} 
\label{c_S}
c(S) &=\sum_{j} (-1)^{P(R^j)}  v(R^j) \prod_{i}\left[-a(D^j_{i})\right]
\\
\label{a_S}
a(S) &=\sum_{j} \prod_{i}\left[-c(A^j_{i})\right]
\end{align}
where $D^j_{i}$ represent the $i^{th}$ individual segment in the $j^{th}$ set in \Verb#list_of_set_of_disjoint_segments#, $A^j_{i}$  represent the $i^{th}$ individual segment in the $j^{th}$ set in \Verb#list_of_set_of_adjacent_segments#, and finally $R^j = S \backslash \bigcup_{i} D^j_{i}$ is the remainder of the of the vertices in the segment $S$ that are not included in any segment $D^j_{i}$ for the $j^{th}$ set in \Verb#list_of_set_of_disjoint_segments#. $v(R^j)$ is the determinant of the hybridization difference matrix of the remainding indices $R^j$.

Note that the sign $ (-1)^{P(R^j)}$ is important and missing from the paper of~\cite{boag_inclusion-exclusion_2018}. This sign is non trivial. The number $P(R^j)$ corresponds to the number of permutations that would be required to reunite all the indices in $R^j$ into a continuous segment at the rightmost position of the segment $S$. This $ (-1)^{P(R^j)}$ is necessary to reproduce the $\operatorname{sgn}(\sigma)$ part of Eq.~\eqref{eq::w_hyb}. 

Let's take an example of a specific segment: the third segment of the first list (shown above), since it is long and non trivial. This segment define the range $[0,6[$.

Note that we need to exclude the \Verb#list_of_set_of_disjoint_segments# that overlap with the last vertex since we need to calculate the fully connected segment (see Boad et al. 2018 for details, for now at least). We use the value $a(S)$ from the previously calculated smaller segments, calculate the sign $ (-1)^{P(R^j)}$ and the determinant $v(R^j)$ of the remainding indices, and thus calculate Eq~\eqref{c_S}.

For the calculation of $a(S)$ (Eq~\eqref{a_S}) we need to refer to the list of set of adjacent segments above: every \Verb#list_of_set_of_adjacent_segments# that have exactly the same range $[0,6[$ must be considered. \subsubsection{Optimizations}

The algorithm presented here already contain one optimization, the first one presented in the paper of ~\cite{boag_inclusion-exclusion_2018}. Indeed, we already expressed everything in terms of adjacent segments only here, in order to simplify the formalism. In this same paper, there is one additonnal optimization is presented, a simple one. We can note that $c(S)$ is always zero for segment of length $2$. In that case, we can simply replace in Eq.~\eqref{eq::hyb_mat} the elements that describe adjacent points (segment of length two) by zero. If we do so, we can ignore every segment of length 2, thus the number of combination of segments is greatly reduced. 

A third optimization is to remove the segment of length $4$ of the type \Verb#x-o-x-o# and  \Verb#o-x-o-x#, as they never can be fully connected, hence will always give zero. The proof is given to the reader, but essentially we just need to enumerate the possible diagrams (only 2), and see that only one hybridization line can cross the split point if it is just before the last vertex.

A fourth optimization is to check, just before calculating a new determinant $v(R^j)$ if any of the previously calculated $D^j_{i}$ are zero. If this is the case, we can just skip the determinant calculation of $v(R^j)$.

\subsubsection{Size of these list}

It is hard to predict the exact size of these list. But we can consider the worst case scenario. Indeed, always alternating vertex \Verb#x-o-x-o-x-o-x-o-...-x-o# will generate the biggest number of segment, thus the biggest lists. The worst case is also when we have no split point (even if in theory, no split point is easier to calculate because the diagram become trivial: just equal to the determinant). I did the exact calculation for this worst case without the small segment optimization (including the segments of size $2$) for simplicity first. In that case, we know the  \Verb#segment_list# size exactly: $k+\sum_{n=1}^{k-1} (2n) = k^2$. The \Verb#list_of_set_of_adjacent_segments# size is exactly: $\sum_{n=1}^{k-1} 2^{n-1}(2n+1) $. The factor $2^{n-1}$ comes from the fact that there is exactly this numer of way to cut an alternating segment with $2n$ vertex (personnal notes: proven with the sum of rows in the Pascal triangle). These two lists are the smallest. The list that is a problem is the  \Verb#list_of_set_of_disjoint_segments#. I did not find a way to calculate the exact upperbound size, but it is quite large.

Of course, removing the segment of size $2$ make these list much smaller. 




\section{Monte Carlo}

\subsection{Updates}
The equation of the propagator (Eq.~\eqref{eq::ordered_order} for CT-HYB Eq.~\eqref{eq::inchworm_order} for inchworm) involve many integrals. One way to calculate this is to use Monte Carlo sampling. 
\begin{align}
U(\tau)=& 
\sum_k^{\infty} \int \prod_i^{2k} \Big( d \tau_{i} \Big)  \sum_{\{a_i\}} 
w_{\mathrm{at}} w_{\mathrm{hyb}}
\label{eq::U_sample}
\end{align}
%
%
where the weights are defined above. The Monte Carlo procedure is the same for ct-hyb and inchworm equations. The insert/remove update check to insert/remove a pair of $d^\dagger$ and $d$ anywhere in the $[0,\tau_{max}]$ interval. The acceptance ratio is then:
\begin{align}
R_{xy}&=\frac{t_{yx}}
             {t_{xy}}
        \frac{\vert\vert \tilde U_y \vert\vert_F}
             {\vert\vert \tilde U_x \vert\vert_F}\\
      &=\left(\frac{n_a \tau_{max} }{k+1} \right)^2
        \frac{\vert\vert w^y_{\mathrm{at}} \vert\vert_F w^y_{\mathrm{hyb}}}
             {\vert\vert w^x_{\mathrm{at}} \vert\vert_F w^x_{\mathrm{hyb}}}
\end{align}
where $\tilde U_x = w_{\mathrm{at}} w_{\mathrm{hyb}}$ for configuration $x$ and $\vert\vert \tilde U_x \vert\vert_F$ is the frobenius norm of $\tilde U_x$. The configuration $x$ is a set of $k$ operators $D_a (\text{with} \; \tau_a), D_b (\text{with} \; \tau_b), \cdots, D_{2k} (\text{with} \; \tau_{2k})$. The times $\tau_\alpha$ must all be different but the orbital indices $\alpha$ can be the same. $k$ is the order of expansion of our Markov chain. $n_a$ is the number of orbitals.

\subsection{Measures}
The Monte Carlo sampling of Eq.~\eqref{eq::U_sample} is:
\begin{align}
U(\tau)
\equiv& 
\sum_x \text{sign}(x)
\frac{w^x_{\mathrm{at}} w^x_{\mathrm{hyb}}}
{\vert\vert w^x_{\mathrm{at}} \vert\vert_F}
\end{align}
where the configuration $x$ updated with the insert/remove of the previous section. Not sure how to describe the sign, but there is a sign here.

Unlike cthyb, there is no normalization with the average sign. Instead, the normalization must be done with order 0. Indeed, we know what is the zeroth order value exactly. 
\begin{align}
\tilde U_0(\tau)
\equiv& 
\begin{cases} 
  e^{-\tau H_{\mathrm{at}}} & \text{with cthyb}
      \\
  U(\tau-\tau_{split}) 
  U(\tau_{split})  & \text{with inchworm}
\end{cases}
\end{align}

We can then normalize the Monte Carlo result by finding the normalization constant, i.e. the ratio between the exact $\tilde U_0(\tau)$ and the Monte Carlo estimate of the zeroth order. For this reason, we sample the zeroth order seperatly in order to determine this normalization. 

It has two important implications. If the zeroth order is badly sampled (if for example the inching process have too big $\tau$ step), the normalization will generate a big error, or in the most extreme case, a not-a-number exception. Also, the error on order 0 is null, by definition. But the error on any other order $k$ is the sum of the error estimate of order $k$ plus the error estimate of order $0$ togheter.

\ZB = \operatorname{Tr_B}[ e^{-\beta H_\text{bath} }  ] = 2
\end{align}

\begin{align}
U(\beta)=& \frac{1}{\ZB} e^{-\beta H} = \frac{\cosh(\beta \theta)+1}{2} I_{2 \times 2}
\end{align}

using trigonometric identity and the serie expansion around $\theta=0$, we obtain:
\begin{align}
U(\beta)=& \cosh^2\bigg(\frac{\beta \theta}{2}\bigg) I_{2 \times 2}
=
\bigg(
1
+\frac{\beta^{2} \theta^{2}}{4}
+\frac{\beta^{4} \theta^{4}}{48}
+\frac{\beta^{6} \theta^{6}}{1440}
+\cdots \bigg)
I_{2 \times 2}
\end{align}

The zeroth order of the expansion is the same as Eq.~\eqref{eq::zeroth_order} (with $H_\text{loc}=0$) and the
first order here is the same as the Eq.~\eqref{eq::first_order}, which is a good self-consistency check. For the same system but with a spin (without hybridization between spins), we have:
\begin{align}
U_{spin}(\beta)= U_{spinless}(\beta) \otimes  U_{spinless}(\beta) = \cosh^4\bigg(\frac{\beta \theta}{2}\bigg) I_{4 \times 4}
=
\bigg(
1
+\frac{\beta^{2} \theta^{2}}{2}
+\frac{5 \beta^{4} \theta^{4}}{48}
+\frac{17 \beta^{6} \theta^{6}}{1440}
+\cdots \bigg)
I_{4 \times 4}
\end{align}
which again is coherent with Eq.~\eqref{eq::zeroth_order} and~\eqref{eq::first_order2}.

\subsection{Finite $\epsilon$}

We proceed first with exact diagonalization as in section~\ref{ED_simple}. For this system we have
\begin{align}
H=
H_\text{loc} + H_\text{bath} + H_\text{hyb}=
\begin{blockarray}{ccccc}
& \ket{00} & \ket{01} & \ket{10} & \ket{11} \\
\begin{block}{c(cccc)}
\bra{00} & 0&0&0&0\\
\bra{01} & 0&0&\theta&0\\
\bra{10} & 0&\theta&\epsilon&0\\
\bra{11} & 0&0&0&\epsilon\\
\end{block}
\end{blockarray}
\end{align}

Now technique is the same, but the result is much more complicated because diagonalization produce a lot of squareroots and such. With the help of sympy, we can find the low expansion order of the propagator, which is:

\begin{align}
  U_{00}(\tau_{m}) 
  &= 1+ \frac{\theta^2}{\epsilon^2} 
  \frac{1}{1+e^{\tau_{m}\epsilon}}
  ( e^{\tau_{m}\epsilon} -\tau_{m}\epsilon - 1)
  + O(\theta^4)
  \\
  U_{11}(\tau_{m}) 
  &= 1+ \frac{\theta^2}{\epsilon^2} 
  \frac{1}{1+e^{-\tau_{m}\epsilon}}
  ( e^{-\tau_{m}\epsilon} -\tau_{m}\epsilon - 1)
  + O(\theta^4)
\end{align}

where $\tau_{m}$ is the $\tau$ at which we want to evaluate the inchworm. The indices are the diagonal block of the propagator. This will be only valid for very small $\theta$.

Now this is the exact propagator from our formalism. We can compare that to the low order expansion of the ct-hyb and the inchworm.

First, ct-hyb:

\begin{align}
  U^0_{00}(\tau_{m}) &= U^0_{11}(\tau_{m}) = e^{-\tau_{m} H_{\mathrm{at}}} = 1

  \int_{0}^{\beta} d \tau_a 
  \int_{\tau_a}^{\beta} d \tau_b
  U^0_{00}(\tau_m-\tau_a) d 
  U^0_{00}(\tau_a-\tau_b) d^\dag
  U^0_{00}(\tau_b)
  w_{hyb}

\end{align}
